\textbf{\color{OliveGreen}{Research}}
& \textbf{Research Project in Combinatorics}, Clemson University \\*
& \href{\MastersProjectURL}{Master's project (PDF)} \\*
& \begin{itemize}
    \item Investigated generalizations of Bulgarian solitaire with
    Dr.\ Svetlana Poznanovi\'{c}.
    \item Used Python to visualize orbits and observed that a configuration
    can return to its original shape before individual cards return to
    their original positions.
    \item Developed a coloring algorithm on Ferrers diagrams that connects
    the movement of labeled cells to permutation cycles and orbit sizes.
    \item Coauthored
    \href{\PreprintURL}{\emph{Bulgarian solitaire with distinguishable cells}},
    available on arXiv and submitted for publication.
\end{itemize} \\
\cvspace

& \textbf{Final-Year Research Project}, University of Peradeniya \\*
& \href{\UndergraduateThesisURL}{Undergraduate thesis (PDF)} \\*
& \begin{itemize}
    \item Studied relationships among Collatz sequences with
    Dr.\ Rajitha Ranasinghe.
    \item Implemented a proposed cryptographic scheme in Python
    using the behavior of Collatz sequences.
    \item Coauthored the resulting paper,
    \emph{Collatz conjecture--driven cryptographic framework} (2026).
\end{itemize} \\
\cvspace

& \textbf{Additional Research Projects}, University of Peradeniya \\*
& \begin{itemize}
    \item \textbf{Cryptographic Scheme Based on Graph Theory}
    (October 2022--2023): Studied snake vertex labelings and supermagic
    coverings of wheel graphs for symmetric encryption. This work led to
    \emph{Symmetric encryption using snake graphs and supermagic covering},
    coauthored with P. G. R. S. Ranasinghe and
    A. M. S. L. K. Athapaththu (2024).

    \item \textbf{Constructive Approach to the Collatz Conjecture}
    (June 2021--2023): Constructed a framework for comparing Collatz
    sequences and studied its behavior on odd positive integers.

    \item \textbf{Novel Cryptographic Scheme Based on the Collatz Conjecture}
    (June 2021--2023): Developed a symmetric encryption and decryption
    algorithm based on Collatz sequences. Presented this work at RESCON 2022.
\end{itemize} \\
